async function runAnalysis() { const pair = document.getElementById('pairSelect').value; const timeframe = parseInt(document.getElementById('tfSelect').value); // Generate dynamic randomized price movement so both CALL and PUT appear const base = 1.0850 + (Math.random() * 0.0020 - 0.0010); const prices = [ base - 0.0004, base - 0.0002, base + 0.0001, base - 0.0003, base + (Math.random() > 0.5 ? 0.0005 : -0.0005) ]; try { const res = await fetch(`${API_BASE}/analyze`, { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ pair, timeframe, prices }) }); const data = await res.json(); const sigText = document.getElementById('signalText'); sigText.textContent = data.signal; sigText.className = `text-3xl font-extrabold ${data.signal === 'CALL' ? 'text-emerald-400' : 'text-rose-400'}`; document.getElementById('priceText').textContent = `Price: ${data.current_price.toFixed(4)} | SMA: ${data.benchmark_sma.toFixed(4)}`; logMessage(`Analyzed ${pair} (${timeframe}s): Signal -> ${data.signal}`); } catch(e) { logMessage("Analysis error: " + e.message); } }
